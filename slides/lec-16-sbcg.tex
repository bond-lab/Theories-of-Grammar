\section[Construction grammar: the same analyses in a different system; the family, and how SBCG differs from HPSG]{Construction Grammar}

\myslide{Construction Grammar (CxG)}

\begin{itemize}
\item A family of grammars based on the idea that
  \begin{itemize}
  \item Knowledge of a language comes from \emp{form and function pairings}
  \item \txx{function} includes meaning, content, or intent 
    (both semantics and pragmatics)
  \item \txx{form} includes phonology, syntax, orthography
  \end{itemize}
\item CxG grew out of generative semantics and cognitive linguistics,
  by researchers such as Charles Fillmore, Paul Kay and George Lakoff
\item Instead of language as a grammar+lexicon, think of it as a
  structured network of families of constructions
\end{itemize}

\myslide{Construction Grammars}
\begin{itemize}
\item \txx{Sign-Based Construction Grammar} (Berkeley Construction Grammar)
  \\  unification-based framework (with computational implementation)
\item \txx{Goldbergian/Lakoffian Construction Grammar}
  \\ psychologically plausible
\item \txx{Radical Construction Grammar}
  \\ syntactic categories, roles, and relations are not universal: 
  \\ they are not only language-specific, but also construction-specific 
\item \txx{Embodied Construction Grammar}
  \\ relates constructions to embodiment and sensorimotor experience
\item \txx{Fluid Construction Grammar}
  \\ learns grammars from the environment (with computational implementation)
\end{itemize}


\myslide{Overview of Differences (SBCG vs.\ HPSG)}

\begin{itemize}
\item Multiple inheritance
\item Signs
\item Grammar rules form a hierarchy --- many more rules
\item Every tree node has its own phonology
\item Many principles become constraints on 
grammar rules
\item The definition of well-formedness is 
simplified
\end{itemize}

\qaslide{What Does Each Lexeme Inherit?}{
\begin{itemize}
\item \lex{continue} is both a \typ{verb-lxm} and an \typ{sr-lxm}.
  What comes from each parent?
  \answer{\\ {\small\emp{From \typ{verb-lxm}: \ft{head} \typ{verb}, the
    defaults for \ft{aux} and \ft{form}, \ft{mode} \val{prop}, a nominal
    first argument.  From \typ{sr-lxm}: the shared tag between the first
    argument and the complement's \ft{spr}, and the complement's
    \ft{index} as the \ft{arg} of its predication.}}}
\item \lex{likely} is an \typ{adj-lxm} and an \typ{sr-lxm}.  What is
  the gain?
  \answer{\\ {\small\emp{Raising is stated once, on \typ{sr-lxm}, and both
    \lex{continue} and \lex{likely} inherit it; in Chapter 12 we had to
    write the raising constraint into \typ{srv-lxm} and would have
    needed it again for adjectives.  The same holds for the clause types:
    \eng{Who do we see?} is at once an \typ{int-cl} and a
    \typ{hd-fill-cx}.}}}
\end{itemize}
}

\section[Everything is a sign: words, rules, principles, phonology and the root condition]{Everything Is a Sign}

\myslide{Saussure's Sign}
\MyLogo{\citet{saussure:1916}}
\begin{minipage}[c]{0.42\textwidth}
\begin{center}
\begin{tikzpicture}
  \node[draw, thick, ellipse, minimum width=62mm, minimum height=42mm] (sign) {};
  \draw[thick] (sign.west) -- (sign.east);
  \node[align=center] at ($(sign.center)+(0,9mm)$) {\emp{signifi\'e}\\ \small the concept};
  \node[align=center] at ($(sign.center)+(0,-9mm)$) {\emp{signifiant}\\ \small the sound image};
  \node[anchor=north, font=\small] at ($(sign.south)+(0,-1mm)$) {the sign: two sides of one sheet};
\end{tikzpicture}
\end{center}
\end{minipage}\hfill
\begin{minipage}[c]{0.55\textwidth}
\begin{itemize}\addtolength{\itemsep}{-0.5ex}
\item For \citet{saussure:1916} a linguistic \txx{sign} is a
  \emp{signified} (a concept) and a \emp{signifier} (a sound image)
  bound together, like the two sides of a sheet of paper.
\item The pairing is \txx{arbitrary}: nothing in the sound tells you
  the meaning, so each sign has to be learned and stored.
\item Signs combine into larger units (\emp{syntagmes}), which are
  signs in their own right.
\end{itemize}
\end{minipage}

\myslide{Our Signs Are Saussure's}
\begin{center}
\scalebox{0.8}{\avm{[\type*{word}
   phon & < \textnormal{Kim} > \\
   syn & [ head & [\type*{noun} agr & 3sing ] ] \\
   sem & [ mode & ref \\ index & i \\
           restr & < [ reln & \prd{name} \\ name & \textnormal{Kim} \\ named & i ] > ] ]}}
\end{center}
\vspace{-2ex}
\begin{itemize}\addtolength{\itemsep}{-1.5ex}
\item \ft{phon} is the signifier and \ft{sem} the signified, held in
  one feature structure: neither exists without the other.
\item The entry is stored in the lexicon because the pairing is
  arbitrary; nothing predicts that \emp{Kim} names Kim.
\item \ft{syn} adds how the sign combines with others, Saussure's
  \emp{syntagmatic relations}; and phrases are signs too, their
  signifier built by the Principle of Order (next slides) and their
  signified by the Semantic Compositionality Principle.
\end{itemize}

\myslide{Words Are Signs}
    \avm{
      < \textnormal{dog}, \ [\type*{word}
      head & \type{noun} \\
      % [\type*{agr-cat} agr & [ per & 3 \\ num & sg ] ]
      val & [ spr & < D > \\ comps & < > ]
      ]>} 

    Becomes:

    \avm{
     \ [\type*{word}
        phon & < \textnormal{dog} > \\
      syn &  [ head & \type{noun} \\
      % [\type*{agr-cat} agr & [ per & 3 \\ num & sg ] ]
      val & [ spr & < D > \\ comps & < > ] ]
      ]} 

\myslide{Rules Are Signs}
    \avm{[\type*{phrase}\\
             val & [ comps & < >\\
                      spr & < >]]\ \ \
$\rightarrow$\ \ \ \ \2\ \ \ \ \HD[ %\type{phrase}\\
			                val & [ comps & < >\\
                                                 spr & < \2 > ]]}

 Becomes:
 
 \avm{
   [\type*{head-spr-cx}\\
     mother & [ val.spr & < > ] \\
     hd-dtr & \2 [ val &  [ spr & < \1 > \\
                              comps & < >  ] ] \\ 
     dtrs & < \1, \2 >\\
     ]}

\myslide{Principles Are Signs}

\begin{itemize}
\item Head Feature Principle   \\
\avm{
  [ mother.syn.head & \1 \\
  hd-dtr.syn.head & \1  ]}
\item Semantic Inheritance Principle \\
  \avm{
  [ mother.sem & [ mode  & \1 \\
                     index & \2 ] \\
  hd-dtr.sem & [ mode  & \1 \\
                     index & \2 ]  ]}
\item Valence Principle \\
  \avm{
  [ mother.syn.val & / \1 \\
  hd-dtr.syn.val & / \1  ]}
\end{itemize} 

\myslide{Phonology Is Built Too: The Principle of Order}
\begin{center}
\avm{\typ{cx}: [ mother & [ phon & \tag{A1} $\oplus \cdots \oplus$ \tag{An} ] \\
                 dtrs & < [ phon & \tag{A1} ], \ldots, [ phon & \tag{An} ] > ]}
\end{center}
\begin{itemize}
\item Every sign has a \ft{phon} list, and a construction's mother
  just concatenates its daughters' lists, in order.
\item So the tree is no longer a separate object: word order, structure
  and meaning are all features of one sign.
\end{itemize}

\qaslide{Where Did the Head Feature Principle Go?}{
\begin{itemize}
\item In Chapter 4 the HFP was a principle applying to every headed
  phrase.  Where is it now?
  \answer{\\ {\small\emp{It is a constraint on the type \typ{hd-cx}
    (headed construction).  Every headed construction, \typ{hd-spr-cx},
    \typ{hd-comp-cx}, \typ{hd-fill-cx}, \ldots, inherits it; the
    non-headed ones (imperative, coordination) simply do not have a
    \ft{hd-dtr}.}}}
\item What is left of ``well-formedness''?
  \answer{\\ {\small\emp{Only this: a structure is well formed if it is the
    \ft{mother} of some instance of some construction.  There are no
    principles that look at trees from outside.}}}
\end{itemize}
}

\myslide{The Root Condition (Start Symbol) Is a Sign}

\avm{

        form & fin ] \\
      val & [ comps & < > \\
        spr  & < > ] \\
      gap & < > ] ]}

\myslide{Sign-Based Construction Grammar}

\begin{itemize}
\item SBCG makes the analyses more uniform
\item Puts information just where it is needed
\item Merges the insights of HPSG and Construction Grammar
\item Most recent work on HPSG is effectively work on SBCG
\end{itemize}
