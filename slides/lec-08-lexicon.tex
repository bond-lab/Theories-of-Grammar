\qaslide{Motivation for the Lexical Hierarchy}{
\begin{itemize}
\item We've streamlined our grammar rules\ldots
  \begin{itemize}
  \item \ldots by stating some constraints as general principles
  \item \ldots and locating lots of information in the lexicon.
  \end{itemize}
\item Our lexical entries currently stipulate a lot of
  information that is common across many entries and
  should be stated only once.  Examples?
  \answer{%
  \begin{itemize}
  \item \emp{Every common noun: \avm{[ head & [\type*{noun} agr & [ per & 3rd ] ] ]},
      \avm{[ spr & < DP > ]}, \avm{[ mode & ref ]}}
  \item \emp{Every verb: \avm{[ head & \type{verb} ]}, \avm{[ mode & prop ]},
      an NP subject first on \ft{arg-st}}
  \item \emp{Every noun and verb: the SHAC}
  \end{itemize}}
\end{itemize}
}

\myslide{The Ideal Lexical Entry}
\begin{itemize}
\item Ideally, particular lexical entries need only give
  phonological form, the semantic contribution,
  and any constraints truly idiosyncratic to the
  lexical entry.
  \begin{itemize}
  \item We don't get quite that far in this book, because of our
    `cutesy' role names (\ft{lover}, \ft{giver}, \ldots): the entry
    still has to say which argument plays which role
  \item Implemented grammars such as the ERG use generic role labels
    (\ft{arg1}, \ft{arg2}), so the mapping from arguments to roles can
    live in the lexical type
  \end{itemize}
\end{itemize}

%%% p7
\myslide{Lexemes and Words (in a General Linguistic Sense)}
\begin{itemize}
\item \txx{Lexeme}: An abstract proto-word which gives rise
  to genuine words. We refer to lexemes by their
  `dictionary form', e.g.\ `the lexeme \eng{run}' or `the
  lexeme \eng{dog}'.
\item \txx{Word}: A particular pairing of form and meaning.
  \eng{Running} and \eng{ran} are different words
\end{itemize}

%%% p8
\qaslide{Lexical Entries, Lexical Types \& Lexical Rules}{
\begin{itemize}
\item \txx{Lexical entries} capture the similarities among \lex{run, runs,
    running,} and \lex{ran}.
\item The \txx{lexical type hierarchy} captures the similarities among
  \lex{run, sleep,} and \lex{laugh}, among those and other verbs like
  \lex{devour} and \lex{hand}, and among those and other words like
  \lex{book}.
  \begin{itemize}
  \item[Q] What do \lex{devour} and \lex{book} have in common?
  \answer{\item[A] \emp{The SHAC: each agrees with its specifier}}
  \end{itemize}
\item \txx{Lexical rules} capture the similarities among \lex{runs, sleeps,
    devours, hands,}\ldots
\end{itemize}
}

%%% p11
\myslide{\val{lexeme}}
\begin{itemize}
\item A \emp{type} in our grammar
\item The root of the sub-hierarchy of all the lexeme types we use in
  defining lexical entries
\item So `a lexeme' loosely refers to the type \val{lexeme} or one of
  its subtypes
\item These types are generalizations about combinations of
  constraints
  \begin{itemize}
  \item e.g.\ the lexeme type \val{pn-lxm} encodes the generalization
    that proper nouns are (usually) singular
  \item the lexeme type \val{stv-lxm} encodes the generalization that
    a transitive verb takes two arguments
  \end{itemize}
\end{itemize}

%%% p12
\myslide{Lexical Entry}
\begin{itemize}
\item A piece of the grammar that we write in the lexicon
  \begin{itemize}
  \item e.g.\ `look, we have \lex{book}, it's of type \val{cntn-lxm},
    and we have \lex{give}, it's of type \val{dtv-lxm}'
  \end{itemize}
\end{itemize}
\begin{center}
\scalebox{0.72}{%
\begin{tabular}{@{}l@{\qquad}l@{}}
\avm{< \textnormal{book} , [\type*{cntn-lxm}
    sem & [ index & $i$ \\
            restr & < [ reln & book \\ inst & $i$ ] > ] ] >}
&
\avm{< \textnormal{give} , [\type*{dtv-lxm}
    sem & [ index & $s$ \\
            restr & < [ reln & give \\ sit & $s$ \\
                        giver & $i$ \\ recipient & $j$ \\ gift & $k$ ] > ] \\
    arg-st & < X$_i$, Y$_j$, Z$_k$ > ] >}
\end{tabular}}
\end{center}

%%% p13
\myslide{Lexical Sequence}
\begin{itemize}
\item A \emp{fully resolved feature structure} that is a pair of some
  orthographic or phonological string with some feature structure of
  type \val{lexeme} or \val{word}
\item There is a type \val{l-sequence} to encode this
  \begin{itemize}
  \item the \ft{input} and \ft{output} features of lexical rules have
    values of type \val{l-sequence}
  \end{itemize}
\end{itemize}
\begin{center}
\begin{small}
\begin{tabular}{|l|l|l|}\hline
\textsc{type} & \textsc{features/constraints} & \textsc{ist} \\ \hline
\val{list} & & \val{feat-struc} \\ \hline
\val{list($\tau$)} & \avm{[ first & $\tau$ \\ rest & \type{list($\tau$)} ]}
  & \val{list} \\ \hline
\val{l-sequence} & \avm{[ first & \type{atom} \\
                         rest & < \type{word} > $|$ < \type{lexeme} > ]}
  & \val{list} \\ \hline
\end{tabular}
\end{small}
\end{center}
\emp{In plain English: an \val{l-sequence} is a two-element list,
  a string followed by one \val{word} or \val{lexeme}.}

%%% p14
\qaslide{Lexical Sequences?}{
\begin{center}
\scalebox{0.62}{%
\avm{< \textnormal{book} , [\type*{cntn-lxm}
  syn & [ head & [\type*{noun}
                  agr & \1 [\type*{3sing} per & 3rd \\ num & sg \\ gend & neut ] \\
                  case & nom ] \\
          val & [ spr & < [ agr & \1 ] > \\ comps & < > \\ mod & < > ] ] \\
  sem & [ mode & ref \\ index & $i$ \\
          restr & < [ reln & book \\ inst & $i$ ] > ] \\
  arg-st & < DP$_i$[ count & + ] > ] >}
\qquad
\avm{< \textnormal{book} , [\type*{cntn-lxm}
  syn & [ head & [\type*{noun}
                  agr & \1 [\type*{plural} per & 3rd \\ num & pl ] \\
                  case & acc ] \\
          val & [ spr & < [ agr & \1 ] > \\ comps & < > \\ mod & < > ] ] \\
  sem & [ mode & ref \\ index & $i$ \\
          restr & < [ reln & book \\ inst & $i$ ] > ] \\
  arg-st & < DP$_i$[ count & + ] > ] >}}
\end{center}
\begin{itemize}
\item[Q] Technically, neither of these is a lexical sequence.  Why?
\answer{\item[A] \emp{Lexical sequences must be \textbf{fully
      resolved}, and these don't fully resolve the DP on the
      \ft{arg-st} and \ft{spr} lists.}}
\end{itemize}
}

%%% p15
\myslide{`Family of Lexical Sequences'}
\begin{itemize}
\item A pairing of orthographic form with some \emp{underspecified}
  feature structure of type \val{lexeme} or \val{word}, which can be
  fully resolved to a number of different lexical sequences
\item The book often refers to lexical entries, for example, as being
  `families of lexical sequences', because they license infinitely
  many fully resolved lexical sequences
\item So the examples on the previous slide are technically all
  families of lexical sequences, because the \ft{arg-st} and \ft{val}
  constituents are not fully specified
\end{itemize}

%%% p17
\myslide{Lexical Rule}
\begin{itemize}
\item A part of the grammar that we write that allows us to derive\ldots
  \begin{itemize}
  \item inflected forms of words, given lexical entries
  \item or additional lexical sequences where the feature structure is
    some subtype of \val{lexeme}
    \begin{itemize}
    \item \ldots which can unify with the \ft{input} feature of
      additional lexical rules until we arrive at a lexical sequence
      pairing a word-form with a feature structure of type \val{word}
    \end{itemize}
  \end{itemize}
\end{itemize}

\myslide{Lexical Rule: An Example}
\begin{center}
3rd-Singular Verb Lexical Rule

\medskip
\avm{[\type*{i-rule}
  input & < \3 , [\type*{verb-lxm} sem & [ restr & \tag{A} ] ] > \\
  output & < F$_{3SG}$(\3) ,
            [ syn & [ head & [ form & fin \\ agr & \type{3sing} ] ] \\
              sem & [ restr & \tag{A} $\oplus$ \ldots ] \\
              arg-st & < [ case & nom ], \ldots > ] > ]}
\end{center}
\emp{In plain English: take a verb lexeme, add the 3rd singular
  ending, and make it a finite word with a nominative subject that
  agrees with it.}

%%% p18
\myslide{Lexical Rule Instantiation}
\begin{itemize}
\item A \emp{fully resolved} feature structure that corresponds to
  one of those rules in the grammar
\item That is, the rule with a particular lexical sequence as its
  \ft{input} and the corresponding lexical sequence as its \ft{output}
  \begin{itemize}
  \item e.g.\ the 3rd-Singular Verb Lexical Rule with
    $\langle$\textnormal{sleep}, \ldots$\rangle$ as \ft{input} and
    $\langle$\textnormal{sleeps}, \ldots$\rangle$ as \ft{output}
  \item one rule in the grammar; one instantiation per word it licenses
  \end{itemize}
\end{itemize}

%%% p22
\qaslide{Default Inheritance}{
Why do we have default inheritance?
\answer{%
\begin{itemize}
\item[A] \emp{Generalizations with exceptions are common:}
  \begin{itemize}
  \item Most nouns in English aren't marked for \ft{case}, but
    pronouns are.
  \item Most verbs in English only distinguish two agreement
    categories (\val{3sing} and \val{non-3sing}), but \lex{be} distinguishes
    more.
  \item Most prepositions in English are transitive, but \lex{here} and
    \lex{there} are intransitive.
  \item Most nominal words in English are 3rd person, but some
    (all of them pronouns) are 1st or 2nd person.
  \item Most proper nouns in English are singular, but some
    (mountain range names, sports team names) are plural.
  \end{itemize}
\end{itemize}
\begin{small}
  Note: defaults are not necessary, and currently implemented grammars
  do not use default inheritance, though some work is looking into
  adding it.
\end{small}}
}

\newcommand{\dftrow}[3]{%
\raggedright #1 &
\raggedright #2 &
\raggedright #3\tabularnewline}
\qaslide{Default Inheritance, Technicalities}{
\vspace*{1em}
\begin{tabular}{@{}p{0.28\textwidth}p{0.28\textwidth}p{0.32\textwidth}@{}}
\dftrow{If a type says \avm{[ arg-st & / < NP > ]},}%
       {and one of its subtypes says \avm{[ arg-st & < > ]},}%
       {\answer{then the \ft{arg-st} value of instances of the subtype
           is \avm{< >}.}}
\end{tabular}
\answer{\par\bigskip
  \emp{In plain English: the `\,/\,' marks a default.  A subtype can
    override it, and the subtype's value wins.}}
}

\qaslide{Default Inheritance, Technicalities}{
\vspace*{1em}
\begin{tabular}{@{}p{0.28\textwidth}p{0.28\textwidth}p{0.32\textwidth}@{}}
\dftrow{If a type says \avm{[ arg-st & / < NP > ]},}%
       {and one of its subtypes says \avm{[ arg-st & < > ]},}%
       {then the \ft{arg-st} value of instances of the subtype is
         \avm{< >}.}
\dftrow{\vspace*{3em}If a type says \avm{[ arg-st & < NP > ]},}%
       {\vspace*{3em}and one of its subtypes says \avm{[ arg-st & < > ]},}%
       {\vspace*{3em}\answer{then this subtype can have no instances,
           since they would have to satisfy contradictory constraints.}}
\end{tabular}
\answer{\par\bigskip
  \emp{In plain English: without the `\,/\,' the constraint is
    inviolable.  Nothing can be both \avm{< NP >} and \avm{< >}, so
    the subtype is empty.}}
}

%%% p29
\qaslide{Default Inheritance, More Technicalities}{
\begin{itemize}
\item If a type says \avm{[ mod & / < S > ]}, and one of its subtypes says
  \avm{[ mod & < [ spr & < NP > ] > ]}, then the \ft{mod} value of
  instances of the subtype is what?
\end{itemize}
\answer{%
\begin{center}
  \avm{[ mod & < [ head & / \type{verb} \\
                   spr & < NP > \\
                   comps & / < > ] > ]}
\end{center}
\begin{itemize}
\item That is, default constraints are `pushed down'
\item \emp{S abbreviates
    \avm{[ head & \type{verb} \\ spr & < > \\ comps & < > ]}, so the
    subtype only overrides the part it mentions (\ft{spr}).  The rest
    of the default survives, still as a default.}
\end{itemize}}
}

%%% p30
\qaslide{Question on Default Inheritance}{
\begin{itemize}
\item[Q] Can a grammar rule override a default
  constraint on a word?
\answer{\item[A] \emp{No. Defaults are all `cached out' in the
  lexicon.}
\item Words as used to build sentences have only
  inviolable constraints.
  \begin{itemize}
  \item i.e.\ nodes of type \val{word} in your tree should never be
    shown with a defeasible constraint (no `\,/\,' in a tree)
  \end{itemize}}
\end{itemize}
}

\myslide{How Lexical Rules Fit In: Well-Formed Tree Structure}
$\Phi$ is a well-formed tree structure according to grammar G
if and only if:
\begin{enumerate}
\item $\Phi$ is a tree structure,
\item the label of $\Phi$'s root node satisfies the constraint
  \avm{[ syn & [ head & [\type*{verb} form & fin ] \\
                 val & [ comps & < > \\ spr & < > ] ] ]} and
\item each local subtree within $\Phi$ is either phrasally licensed
  or lexically licensed.
\end{enumerate}
\emp{A sentence is a finite clause with nothing missing, where every
  branching is licensed by a rule and every leaf by the lexicon.}

\myslide{How Lexical Rules Fit In: Lexical Sequences}
$\langle\omega, \phi\rangle$ is a lexical sequence if and only if
$\omega$ is a phonological form (an atom), $\phi$ is a feature
structure, and either:
\begin{enumerate}
\item G contains some lexical entry $\langle d_1, d_2\rangle$
  such that $\omega$ satisfies $d_1$ and $\phi$ satisfies $d_2$, or
\item there is some lexical rule instantiation licensed by G
  (a feature structure of type \val{l-rule}) whose \ft{output}
  value is $\langle\omega, \phi\rangle$.
\end{enumerate}
\emp{A form paired with a feature structure is a lexical sequence if
  it comes straight from a lexical entry, or is the output of a
  lexical rule.  The second clause is recursive: the rule's input
  must itself be a lexical sequence, so rules can feed each other.}

\myslide{How Lexical Rules Fit In: Lexical Licensing}
\begin{minipage}[c]{0.7\textwidth}
A word structure of the form on the right is licensed if and only if:
\begin{enumerate}
\item $\langle\omega, \phi\rangle$ is a lexical sequence, where
  $\phi$ is of type \val{word},
\item (Case Constraint:) an outranked NP is [\ft{case} acc], and
\item $\phi$ satisfies the Binding Theory.
\end{enumerate}
\end{minipage}%
\begin{minipage}[c]{0.25\textwidth}
\begin{center}
\ctree{[{$\phi$} [{$\omega$}]]}
\end{center}
\end{minipage}

\medskip
\emp{Only lexical sequences of type \val{word} can appear in trees.
  Lexemes have to pass through at least one lexical rule to become
  words: that is what the inflectional rules are for.}

\myslide{Parsimony \& Plausibility}
\begin{itemize}
\item Lexical rules capture \textbf{productive}
  generalizations.
  \begin{itemize}
  \item When a new word enters the lexicon, it is assigned a lexical
    type, and we expect every lexical rule that applies to that type
    to apply to the new word too.
  \end{itemize}
\item There may be some `precompiling'
  going on as well.
  \begin{itemize}
  \item Psycholinguistically, very frequent words may just be stored
    ready-made rather than derived each time.
  \end{itemize}
\end{itemize}

\myslide{Three Subtypes of \textit{l-rule}}
\begin{center}
\begin{small}
  \begin{forest}
    cat, for tree={font=\itshape, l sep=3mm}
    [{l-rule} [{i-rule}] [{d-rule}] [{pi-rule}]]
  \end{forest}

  \medskip
  \val{l-rule} :
  \avm{[ input & \type{l-sequence} < X , [ sem & / \2 ] > \\
         output & \type{l-sequence} < Y , [ sem & / \2 ] > ]}

  \medskip
  \val{i-rule} :
  \avm{[ input & < X , [\type*{lexeme}
                         syn & \3 \\
                         arg-st & \tag{A} ] > \\
         output & < Y , [\type*{word}
                          syn & \3 \\
                          arg-st & \tag{A} ] > ]}
  \qquad
  \val{d-rule} :
  \avm{[ input & < X , [\type*{lexeme}
                         syn & / \3 ] > \\
         output & < Y , [\type*{lexeme}
                          syn & / \3 ] > ]}
\end{small}
\end{center}
\begin{small}
\emp{\val{l-rule}: the output keeps the input's meaning, by default.
  \val{i-rule}: the output is a \val{word} with the same \ft{syn} and
  \ft{arg-st}, no exceptions.
  \val{d-rule}: the output is another \val{lexeme} with the same
  \ft{syn}, by default.}
\end{small}

\myslide{Practicalities --- Applying Lexical Rules}
\begin{itemize}
\item \ft{input} is a family of lexical sequences.
\item \ft{output} is another family of lexical sequences.
  \begin{itemize}
  \item \ldots usually a smaller family
  \item \ldots usually a disjoint one
  \end{itemize}
\item The only differences between the families are those
  stipulated in the rule (or the rule's type).
\item Similarities are handled by the constraints on \val{l-rule}
  and its subtypes.
\item If we've written the LRs correctly, nothing is left
  underconstrained.
\end{itemize}

\myslide{Practicalities --- Writing Lexical Rules}
\begin{itemize}
\item Determine the type of the LR.
\item Determine the class of possible inputs.
\item Determine what should change.
  \begin{small}
  \begin{itemize}
  \item If \ft{input} and \ft{output} values are identified (by default or otherwise) and
    only \ft{output} value is mentioned, then\ldots
    \\ \emp{information is added.}  e.g.\ Plural Noun LR
    \\[1ex] (Lexical sequences incompatible with that value are not possible inputs)
  \item If \ft{input} and \ft{output} values are identified by default, but different values
    are given on the \ft{input} and \ft{output} of the rule, then\ldots
    \\ \emp{information is changed.}  e.g.\ Agent Nominalization LR
  \item If \ft{input} and \ft{output} values are identified by an inviolable constraint, but
    different values are given on the \ft{input} and \ft{output} of the rule, then\ldots
    \\ \emp{there is no well-formed input: the rule can never apply.}
  \end{itemize}
  \end{small}
\end{itemize}

\qaslide{Constant Lexemes}{
\begin{itemize}
\item What kinds of words are constant lexemes
  in our grammar?
  \answer{\\ \emp{Adjectives, adverbs, prepositions, determiners,
      conjunctions: words that do not inflect in English.}}
\item Why do we need a rule for these words?
  \answer{\\ \emp{They need to go `from' \val{lexeme} to \val{word},
      and only words appear in trees.}}
\item What would be an alternative analysis?
  \answer{\\ \emp{Put the sub-hierarchy for constant lexemes under
      \val{word} instead of under \val{lexeme}.}}
\end{itemize}
}

%%% p38--39
\qaslide{Constant Lexeme LR}{
\begin{center}
  \avm{[\type*{i-rule}
    input & < \1 , \type{const-lxm} > \\
    output & [ first & \1 ] ]}
  \qquad i.e.\qquad
  \avm{[\type*{i-rule}
    input & < \1 , \type{const-lxm} > \\
    output & < \1 , [ ] > ]}
\end{center}
\emp{The form does not change, and the rule says nothing else.  The
  type \val{i-rule} does the work: it copies \ft{syn} and \ft{arg-st}
  and makes the output a \val{word}.}
\begin{itemize}
\item What keeps this from applying to, say, verb
  lexemes?
  \answer{\\ \emp{The input must be a \val{const-lxm}.  Verb lexemes
      are \val{infl-lxm}s, a different branch of the hierarchy.}}
\item Why is this an \val{i-rule}?
  \answer{\\ \emp{It takes a \val{lexeme} to a \val{word}, which is
      what inflectional rules do, even though nothing is inflected.}}
\end{itemize}
}

\myslide{Other Lexical Rules (So Far)}
\begin{itemize}
\item Inflectional (\val{i-rule}): Singular Noun, 3rd-Singular Verb,
  Non-3rd-Singular Verb, Past-Tense Verb, Base Form
\item Derivational (\val{d-rule}): Present Participle, Past
  Participle, Agent Nominalization
\item Each one is in the grammar summary in Chapter 9.  On the next
  slides, read each rule and ask: what is the input, what changes, and
  where does the rest come from?
\end{itemize}

\myslide{Singular Noun Lexical Rule}
\begin{center}
  \avm{[\type*{i-rule}
    input & < \1 , \type{cn-lxm} > \\
    output & < \1 , [ syn & [ head & [ agr & [ num & sg ] ] ] ] > ]}
\end{center}
\emp{Any common noun lexeme, count or mass, gives a singular word
  with the same form.  Compare the Plural Noun LR, whose input is
  only \val{cntn-lxm}: mass nouns have no plural.}

\myslide{3rd-Singular Verb Lexical Rule}
\begin{center}
  \avm{[\type*{i-rule}
    input & < \3 , [\type*{verb-lxm} sem & [ restr & \tag{A} ] ] > \\
    output & < F$_{3SG}$(\3) ,
              [ syn & [ head & [ form & fin \\ agr & \type{3sing} ] ] \\
                sem & [ restr & \tag{A} $\oplus$ \ldots ] \\
                arg-st & < [ case & nom ], \ldots > ] > ]}
\end{center}
\emp{Add \textit{-s}; the word is finite and 3rd singular, its
  subject (the first argument) is nominative, and the \ft{restr}
  gets something extra (the tense).}

\myslide{Non-3rd-Singular Verb Lexical Rule}
\begin{center}
  \avm{[\type*{i-rule}
    input & < \1 , [\type*{verb-lxm} sem & [ restr & \tag{A} ] ] > \\
    output & < \1 ,
              [ syn & [ head & [ form & fin \\ agr & \type{non-3sing} ] ] \\
                sem & [ restr & \tag{A} $\oplus$ \ldots ] \\
                arg-st & < [ case & nom ], \ldots > ] > ]}
\end{center}
\emp{The same, but the form stays as it is and the agreement is
  anything but 3rd singular: \eng{I/you/we/they walk}.}

\myslide{Past-Tense Verb Lexical Rule}
\begin{center}
  \avm{[\type*{i-rule}
    input & < \3 , [\type*{verb-lxm} sem & [ restr & \tag{A} ] ] > \\
    output & < F$_{PAST}$(\3) ,
              [ syn & [ head & [ form & fin ] ] \\
                sem & [ restr & \tag{A} $\oplus$ \ldots ] \\
                arg-st & < [ case & nom ], \ldots > ] > ]}
\end{center}
\emp{Add \textit{-ed}; finite, nominative subject, but no agreement
  constraint at all: \eng{I/you/she/they walked}.}

\myslide{Base Form Lexical Rule}
\begin{center}
  \avm{[\type*{i-rule}
    input & < \1 , \type{verb-lxm} > \\
    output & < \1 , [ syn & [ head & [ form & base ] ] ] > ]}
\end{center}
\emp{Same form, [\ft{form} base]: not finite, so no nominative
  subject and no agreement.  This is the verb in \eng{Eat!}, \eng{to
    eat} and \eng{will eat}.}

\myslide{Present Participle Lexical Rule}
\begin{center}
  \avm{[\type*{d-rule}
    input & < \3 , [\type*{verb-lxm} sem & [ restr & \tag{A} ] \\
                                    arg-st & \tag{B} ] > \\
    output & < F$_{PRP}$(\3) ,
              [\type*{part-lxm}
                syn & [ head & [ form & prp ] ] \\
                sem & [ restr & \tag{A} $\oplus$ \ldots ] \\
                arg-st & \tag{B} ] > ]}
\end{center}
\emp{A \val{d-rule}: \lex{eat} gives the lexeme \lex{eating}, of
  type \val{part-lxm}, with the same arguments.  \val{part-lxm} is a
  \val{const-lxm}, so the Constant Lexeme LR then makes it a word.}

\myslide{Past Participle Lexical Rule}
\begin{center}
  \avm{[\type*{d-rule}
    input & < \3 , [\type*{verb-lxm} sem & [ restr & \tag{A} ] \\
                                    arg-st & \tag{B} ] > \\
    output & < F$_{PSP}$(\3) ,
              [\type*{part-lxm}
                syn & [ head & [ form & psp ] ] \\
                sem & [ restr & \tag{A} $\oplus$ \ldots ] \\
                arg-st & \tag{B} ] > ]}
\end{center}
\emp{The same, with [\ft{form} psp]: \lex{eat} gives \lex{eaten}, as
  in \eng{I have eaten}.}

\myslide{Agent Nominalization Lexical Rule}
\begin{center}
  \avm{[\type*{d-rule}
    input & < \2 , [\type*{verb-lxm} sem & [ index & $s$ ] \\
                                    arg-st & < X$_i$ , NP$_j$ > ] > \\
    output & < F$_{\textit{-er}}$(\2) ,
              [\type*{cntn-lxm}
                sem & [ index & $i$ ] \\
                arg-st & < Y ( , PP$_j$[ form & of ] ) > ] > ]}
\end{center}
\emp{Add \textit{-er} to a strictly transitive verb lexeme.  The
  result is a count noun lexeme whose index is the verb's subject
  index $i$: a \lex{driver} is the one who drives.  The verb's object
  $j$ can only appear as an \eng{of}-PP: \eng{a builder of bridges}.}

\qaslide{\ft{arg-st} \& ARP}{
\begin{itemize}
\item Given the ARP, what do we need to specify
  about the valence properties of words?
  \answer{\\ \emp{Only the length of the \ft{spr} list, and the SHAC
      already does that: nouns and verbs have exactly one specifier.
      The ARP then fills \ft{spr} and \ft{comps} from \ft{arg-st}.}}
\item Why isn't the ARP a constraint on the type
  \val{lexeme}?
  \answer{\\ \emp{Some derivational rules change the argument
      structure (e.g.\ Agent Nominalization).  If \ft{spr} and
      \ft{comps} were already linked to \ft{arg-st} in the lexeme,
      those rules would have to change all three.}}
\end{itemize}
}

\myslide{Review: Lexical Terminology}
\begin{small}
\begin{itemize}
\item \txx{Lexeme}: a type in our grammar; its subtypes state
  generalizations about combinations of constraints
\item \txx{Lexical entry}: what we write in the lexicon; it licenses
  a family of lexical sequences
\item \txx{Lexical sequence}: a form paired with a fully resolved
  feature structure of type \val{lexeme} or \val{word}
\item \txx{Family of lexical sequences}: a form paired with an
  underspecified feature structure, resolvable in many ways
\item \txx{Lexical rule}: what we write to derive new lexical
  sequences from old ones
\item \txx{Lexical rule instantiation}: a fully resolved \val{l-rule}:
  one rule, one input, one output
\item \txx{Word structure}: the bottom of the tree: a form with a
  fully resolved \val{word} feature structure over it
\end{itemize}
\end{small}

\myslide{P1: Arguments in Japanese}
\MyLogo{Based on \citet[Chapter 8, Problem 6]{sag:wasow:bender:2003}}
As noted in Chapter 2, Japanese word order differs from English in a
number of ways, including the fact that it is a `Subject-Object-Verb'
(SOV) language.  The next two slides give some relevant examples. In
the glosses, `{\sc nom}', `{\sc acc}', and `{\sc dat}' stand for
nominative, accusative, and dative case, respectively.  (Note that
Japanese has one more case --- dative --- than English does.  This
doesn't have any important effects on the analysis; it merely requires
that we posit one more possible value of CASE for Japanese than for
English.)\footnote{The examples marked with `*' here are unacceptable
  with the indicated meanings.  Some of these might be well-formed
  with some other meaning of no direct relevance; others might be
  well-formed with special intonation that we will ignore for present
  purposes.}

\myslide{P1: Arguments in Japanese --- Data}
\begin{exe}
  \ex\label{jp:first} %一人 の 男 が その 本 を 読んだ 。
  \gll Hitorino  otoko-ga sono  hon-o yonda. \\
 one  man-{\sc nom} that  book-{\sc acc} read.{\sc pst} \\
\trans {`One man read that book.'}

[cf.\ *Yonda hitorino otoko-ga sono hon-o. \\ {*}Hitorino otoko-ga yonda
sono hon-o. \\ {*}Otoko-ga hitorino sono hon-o yonda. \\ {*}Hitorino
otoko-ga hon-o sono yonda. \\ {*}Hitorino otoko-ni/-o sono hon-o
yonda. \\ {*}Hitorino otoko-ga sono hon-ga/-ni yonda.]
\end{exe}

[cf.\ *Yonda Hanako-ga hon-o. \\ {*}Hanako-ga yonda hon-o. \\
{*}Hanako-ni/-o hon-o yonda. \\ {*}Hanako-ga hon-ni/-ga yonda.]
\end{exe}

\myslide{P1: Arguments in Japanese --- Tasks}
As the contrasting ungrammatical examples show, the verb must appear
in final position in Japanese. In addition, we see that verbs select
for NPs of a particular case, much as in English. In the following
tasks, assume that the nouns and verbs of Japanese are inflected
words, derived by lexical rule from the appropriate lexemes.

\begin{itemize}\addtolength{\itemsep}{1ex}
\item[A.] Write Head-Specifier and Head-Complement Rules for

they different (if at all) from the Head-Specifier and Head-Complement 
Rules for English?
\end{itemize}
